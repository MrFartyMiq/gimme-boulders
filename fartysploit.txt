local function formatNumber(num)
	if not num or num <= 0 then return "0" end
	if num >= 1000000000 then
		return string.format("%.1fB", num / 1000000000):gsub("%.0B", "B")
	elseif num >= 1000000 then
		return string.format("%.1fM", num / 1000000):gsub("%.0M", "M")
	elseif num >= 1000 then
		return string.format("%.1fk", num / 1000):gsub("%.0k", "k")
	else
		return tostring(math.floor(num))
	end
end

local function formatWeight(kg)
	if not kg or kg <= 0 then return "" end
	if kg >= 1000 then
		local tons = kg / 1000
		local formatted = string.format("%.1f", tons):gsub("%.0$", "")
		return (formatted == "1" and "1 ton" or formatted .. " tons")
	else
		local formatted = string.format("%.1f", kg):gsub("%.0$", "")
		return formatted .. "kg"
	end
end

local function getNocturniteRGB()
	local customHue = 0.70 + (math.sin(tick() * 1.5) * 0.125 + 0.125)
	return Color3.fromHSV(customHue, 1, 1)
end

local function getType(model)
	for _, v in ipairs(model:GetDescendants()) do
		if v:IsA("BasePart") then
			local b = v.Name:match("^(.-)_cell")
			if b and colors[b] then return b end
		end
	end
end

local function removeESP(model)
	if objects[model] then
		for _, v in ipairs(objects[model].parts) do if v then v:Destroy() end end
		if objects[model].name then objects[model].name:Destroy() end
		if objects[model].beam then objects[model].beam:Destroy() end
		if objects[model].a0 then objects[model].a0:Destroy() end
		if objects[model].a1 then objects[model].a1:Destroy() end
		objects[model] = nil
	end
	scanned[model] = nil
end

local function addESP(model)
	if scanned[model] or not model:IsA("Model") then return end
	
	local bType = getType(model)
	if not bType then return end
	scanned[model] = true

	local main = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart", true)
	if not main then return end

	local col = (bType == "Nocturnite") and getNocturniteRGB() or colors[bType]
	local data = {type = bType, main = main, parts = {}}

	for _, p in ipairs(model:GetDescendants()) do
		if p:IsA("BasePart") then
			local box = Instance.new("BoxHandleAdornment")
			box.Adornee = p
			box.Size = p.Size + Vector3.new(0.2, 0.2, 0.2)
			box.Color3 = col
			box.Transparency = 1
			box.AlwaysOnTop = true
			box.ZIndex = 10
			box.Parent = espFolder
			table.insert(data.parts, box)
		end
	end

	local tag = Instance.new("BillboardGui")
	tag.Adornee = main
	tag.Size = UDim2.new(0, 140, 0, 35)
	tag.StudsOffset = Vector3.new(0, 10, 0)
	tag.AlwaysOnTop = true
	tag.Enabled = false
	tag.Parent = espFolder

	local txt = Instance.new("TextLabel")
	txt.Size = UDim2.new(1, 0, 1, 0)
	txt.BackgroundTransparency = 1
	txt.Text = bType
	txt.TextSize = 16
	txt.Font = Enum.Font.GothamBold
	txt.TextColor3 = Color3.fromRGB(255, 255, 255)
	txt.TextStrokeTransparency = 0.2
	txt.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	txt.Parent = tag
	data.name = tag

	local a1 = Instance.new("Attachment")
	a1.Parent = main
	data.a1 = a1

	local beam = Instance.new("Beam")
	beam.Color = ColorSequence.new(col)
	beam.Width0 = 0.08
	beam.Width1 = 0.08
	beam.FaceCamera = true
	beam.Enabled = false
	beam.Parent = espFolder
	data.beam = beam

	objects[model] = data

	model.Destroying:Connect(function()
		removeESP(model)
	end)
end

-- --stat puller--
local function getCrystalStats(inst)
	local val = 0
	local luckBoost = 0
	local weight = 0

	for _, v in ipairs(inst:GetDescendants()) do
		if v:IsA("TextLabel") and v.Text ~= "" then
			local text = v.Text:gsub("<[^>]->", "")

			local luckMatch = text:match("Luck:%s*%+?(%d+%.?%d*)%%") or text:match("%+(%d+%.?%d*)%%")
			if luckMatch then
				local num = tonumber(luckMatch)
				if num then luckBoost = math.max(luckBoost, num) end
			end

			local weightMatch = text:match("(%d+%.?%d*)%s*[kK][gG]")
			if weightMatch then
				local num = tonumber(weightMatch)
				if num then weight = math.max(weight, num) end
			end

			local valMatchM = text:match("%$%s*(%d+%.?%d*)%s*[mM]")
			if valMatchM then
				local num = tonumber(valMatchM)
				if num then val = math.max(val, num * 1000000) end
			end
		end
	end

	if luckBoost == 0 or val == 0 or weight == 0 then
		local baseLuck = 0
		local baseVal = 0

		for _, v in ipairs(inst:GetDescendants()) do
			if v:IsA("ValueBase") then
				local n = v.Name:lower()
				if type(v.Value) == "number" then
					if n:find("luck") or n:find("boost") then baseLuck = math.max(baseLuck, v.Value) end
					if n:find("val") or n:find("price") or n:find("worth") then baseVal = math.max(baseVal, v.Value) end
					if n:find("kg") or n:find("weight") then weight = math.max(weight, v.Value) end
				end
			end
		end

		local attrs = inst:GetAttributes()
		for name, attr in pairs(attrs) do
			if type(attr) == "number" then
				local n = name:lower()
				if n:find("luck") or n:find("boost") then baseLuck = math.max(baseLuck, attr) end
				if n:find("val") or n:find("price") or n:find("worth") then baseVal = math.max(baseVal, attr) end
				if n:find("kg") or n:find("weight") then weight = math.max(weight, attr) end
			end
		end

		if luckBoost == 0 then
			luckBoost = (baseLuck > 0 and baseLuck <= 20) and (baseLuck * 100) or baseLuck
		end
		if val == 0 then val = baseVal end
	end

	return val, luckBoost, weight
end

-- --scanner--
local function scanCrystals()
	local bestValModel, bestLuckModel = nil, nil
	local maxVal, maxLuck = -1, -1

	for _, v in ipairs(workspace:GetDescendants()) do
		if (v:IsA("Model") or v:IsA("BasePart")) and v:IsDescendantOf(workspace) then
			local name = v.Name:lower()
			local isPlotOrBase = name:find("plot") or name:find("base") or name:find("tycoon") or name:find("house") or name:find("stand")
			local isCharacter = lp.Character and v:IsDescendantOf(lp.Character)
			
			if not isPlotOrBase and not isCharacter then
				local isCrystal = name:find("crystal") or name:find("gem") or name:find("ore") or name:find("infinitra")
				if isCrystal then
					local val, luck, weight = getCrystalStats(v)
					if val > maxVal then
						maxVal = val
						bestValModel = v
					end
					if luck > maxLuck then
						maxLuck = luck
						bestLuckModel = v
					end
				end
			end
		end
	end

	return bestValModel, bestLuckModel
end

local function clearSingleCrystalESP(targetKey)
	local entry = crystalESP[targetKey]
	if entry.box then entry.box:Destroy() entry.box = nil end
	if entry.name then entry.name:Destroy() entry.name = nil end
	if entry.beam then entry.beam:Destroy() entry.beam = nil end
	if entry.a0 then entry.a0:Destroy() entry.a0 = nil end
	if entry.a1 then entry.a1:Destroy() entry.a1 = nil end
	entry.model = nil
end

local function updateCrystalESP(targetKey, targetModel, color, labelText)
	local entry = crystalESP[targetKey]
	if not targetModel or not targetModel:IsDescendantOf(workspace) then
		clearSingleCrystalESP(targetKey)
		return
	end

	local main = targetModel:IsA("Model") and (targetModel.PrimaryPart or targetModel:FindFirstChildWhichIsA("BasePart", true)) or targetModel
	if not main then clearSingleCrystalESP(targetKey) return end

	local val, luck, weight = getCrystalStats(targetModel)
	local weightStr = formatWeight(weight)
	local weightTag = (weightStr ~= "" and (" | " .. weightStr) or "")

	local statDisplay = ""
	if targetKey == "value" then
		statDisplay = " [$" .. formatNumber(val) .. weightTag .. "]"
	elseif targetKey == "luck" then
		statDisplay = " [+" .. math.floor(luck) .. "%" .. weightTag .. "]"
	end

	local isBoth = (cachedBestVal == cachedBestLuck) and (cachedBestVal == targetModel)
	if isBoth then
		if targetKey == "luck" then
			clearSingleCrystalESP("luck")
			return
		else
			labelText = "Most Valuable & Lucky Crystal"
			statDisplay = " [$" .. formatNumber(val) .. weightTag .. " | +" .. math.floor(luck) .. "%]"
			color = Color3.fromRGB(255, 140, 0)
		end
	end

	local fullLabel = labelText .. statDisplay

	if entry.model ~= targetModel then
		clearSingleCrystalESP(targetKey)
		entry.model = targetModel

		local box = Instance.new("BoxHandleAdornment")
		box.Adornee = main
		box.Size = main.Size + Vector3.new(0.4, 0.4, 0.4)
		box.Color3 = color
		box.Transparency = 0.5
		box.AlwaysOnTop = true
		box.ZIndex = 10
		box.Parent = espFolder
		entry.box = box

		local tag = Instance.new("BillboardGui")
		tag.Adornee = main
		tag.Size = UDim2.new(0, 320, 0, 35)
		tag.StudsOffset = Vector3.new(0, 8, 0)
		tag.AlwaysOnTop = true
		tag.Parent = espFolder

		local txt = Instance.new("TextLabel")
		txt.Name = "Label"
		txt.Size = UDim2.new(1, 0, 1, 0)
		txt.BackgroundTransparency = 1
		txt.Text = fullLabel
		txt.TextSize = 16
		txt.Font = Enum.Font.GothamBold
		txt.TextColor3 = color
		txt.TextStrokeTransparency = 0
		txt.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
		txt.Parent = tag
		entry.name = tag

		local a1 = Instance.new("Attachment")
		a1.Parent = main
		entry.a1 = a1

		local beam = Instance.new("Beam")
		beam.Color = ColorSequence.new(color)
		beam.Width0 = 0.12
		beam.Width1 = 0.12
		beam.FaceCamera = true
		beam.Parent = espFolder
		entry.beam = beam
	else
		if entry.name and entry.name:FindFirstChild("Label") then
			entry.name.Label.Text = fullLabel
		end
	end

	local isEnabled = (settings.mode == "Crystals") and ((targetKey == "value" and settings.topValue) or (targetKey == "luck" and settings.topLuck))
	if entry.box then entry.box.Transparency = (isEnabled and settings.highlight) and 0.4 or 1 end
	if entry.name then entry.name.Enabled = isEnabled and settings.names end

	local char = lp.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if entry.beam then
		if isEnabled and settings.lines and root and entry.a1 then
			if not entry.a0 or entry.a0.Parent ~= root then
				if entry.a0 then entry.a0:Destroy() end
				local a0 = Instance.new("Attachment")
				a0.Parent = root
				entry.a0 = a0
			end
			entry.beam.Attachment0 = entry.a0
			entry.beam.Attachment1 = entry.a1
			entry.beam.Enabled = true
		else
			entry.beam.Enabled = false
		end
	end
end

		b.Text = txt .. ": " .. (enabled and "ON" or "OFF")
		b.TextColor3 = enabled and Color3.fromRGB(120, 255, 150) or Color3.fromRGB(200, 200, 210)
		callback(enabled)
	end)
	return b
end

-- Real-time frame loop
local lastUpdate = 0

RunService.RenderStepped:Connect(function()
	if tick() - lastUpdate < 0.1 then
		return
	end
	lastUpdate = tick()

	local char = lp.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local dynamicNocturniteColor = getNocturniteRGB()

	for model, data in pairs(objects) do
		local isBoulderMode = (settings.mode == "Boulders")
		local active = isBoulderMode and (settings.boulder == "All" or data.type == settings.boulder)
		local currentCol = (data.type == "Nocturnite") and dynamicNocturniteColor or colors[data.type]

		for _, box in ipairs(data.parts) do
			if box then
				box.Color3 = currentCol
				box.Transparency = (active and settings.highlight) and 0.55 or 1
			end
		end

		if data.name then
			data.name.Enabled = active and settings.names
		end

		if data.beam then
			data.beam.Color = ColorSequence.new(currentCol)
			if active and settings.lines and root then
				if not data.a0 or data.a0.Parent ~= root then
					if data.a0 then data.a0:Destroy() end
					local a0 = Instance.new("Attachment")
					a0.Parent = root
					data.a0 = a0
				end
				data.beam.Attachment0 = data.a0
				data.beam.Attachment1 = data.a1
				data.beam.Enabled = true
			else
				data.beam.Enabled = false
			end
		end
	end

	if settings.mode == "Crystals" then
		updateCrystalESP("value", cachedBestVal, Color3.fromRGB(255, 220, 0), "Most Valuable Crystal")
		updateCrystalESP("luck", cachedBestLuck, Color3.fromRGB(0, 255, 100), "Most Lucky Crystal")
	else
		clearSingleCrystalESP("value")
		clearSingleCrystalESP("luck")
	end
end)
